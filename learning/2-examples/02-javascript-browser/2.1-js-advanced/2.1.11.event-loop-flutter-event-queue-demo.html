// worker.mjs（Worker文件，不在主页面执行）
self.onmessage = event => {
  const total = event.data.reduce((sum, n) => sum + n, 0);
  self.postMessage({ total, hasDOM: typeof document !== "undefined" });
};

// main.mjs（通过HTTP加载；worker.mjs在同目录）
const worker = new Worker(new URL("./worker.mjs", import.meta.url), { type: "module" });
worker.onmessage = event => { console.log(event.data); worker.terminate(); };
worker.postMessage([1, 2, 3]); // {total:6, hasDOM:false}
